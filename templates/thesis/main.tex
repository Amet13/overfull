% !TEX TS-program = lualatex
%
% Thesis template — Overfull (https://github.com/amet13/overfull)
% Build:  make build        (or: latexmk)       -> thesis.pdf
% Check:  make check        (or: overfull check)
%
% \DocumentMetadata turns on PDF tagging and declares PDF/UA-2 (accessible PDF)
% conformance. Keep it before \documentclass, as the first command.
%   lang          document language, recorded in the PDF for screen readers.
%                 It does not choose hyphenation patterns: LaTeX hyphenates
%                 American English by default. For another language also
%                 load babel after \documentclass, e.g. \usepackage[ngerman]{babel},
%                 and set lang = de-DE here.
%   pdfstandard   ua-2 = PDF/UA-2 (ISO 14289-2), checked by veraPDF
%   pdfversion    PDF/UA-2 requires PDF 2.0
%   tagging       structure tags for headings, lists, tables, figures, links
%   math/setup    mathml-SE: formulas become MathML structure (LuaLaTeX)
\DocumentMetadata{
  lang        = en,
  pdfstandard = ua-2,
  pdfversion  = 2.0,
  tagging     = on,
  tagging-setup = {math/setup=mathml-SE},
}
% The class is article, so the top level is \section (one per chapter, each
% on a new page); there is no \chapter.
% 14 pt body text, the 1.27 cm paragraph indent (preamble/layout.tex) and the
% uppercase headings (preamble/headings.tex) follow one regional convention,
% common in Eastern Europe. Many universities ask for 12 pt instead: change
% fontsize=14pt to 12pt below, and follow your university's rules.
\documentclass[a4paper]{article}
\usepackage[fontsize=14pt]{fontsize}

\input{metadata}               % title, author, keywords ...
\input{preamble/fonts}         % Times New Roman, math fonts
\input{preamble/layout}        % margins, spacing, page numbers, lists
\input{preamble/headings}      % section headings, table of contents
\input{preamble/floats}        % figures, tables, plots, code listings
\input{preamble/references}    % links, \cref, bibliography
\input{preamble/metadata-pdf}  % PDF title/author from metadata.tex
\input{preamble/titlepage}     % \thesistitlepage

\begin{document}

% ---- Front matter: roman page numbers (i, ii, ...) ------------------------
% The title page is page i but shows no number.
\pagenumbering{roman}
\thesistitlepage

\input{chapters/declaration}
\input{chapters/abstract}
\input{chapters/acknowledgments}
\input{chapters/ai-disclosure}

\tableofcontents

\clearpage
\phantomsection
\addcontentsline{toc}{section}{\listfigurename}
\listoffigures

\clearpage
\phantomsection
\addcontentsline{toc}{section}{\listtablename}
\listoftables

\input{chapters/abbreviations}

% ---- Main matter: arabic page numbers from 1 --------------------------------
\clearpage
\pagenumbering{arabic}

\input{chapters/introduction}
\input{chapters/01-problem}
\input{chapters/02-literature}
\input{chapters/03-analysis}
\input{chapters/04-design}
\input{chapters/05-implementation}
\input{chapters/06-experiments}
\input{chapters/07-results}
\input{chapters/conclusion}

% End of the main matter. `overfull check` counts pages up to this label
% when a page limit is configured in overfull.yml. Keep it here.
\label{overfull:main-end}

\clearpage
\phantomsection
\addcontentsline{toc}{section}{Bibliography}
\printbibliography[title=Bibliography]

% ---- Appendices: \appendixsection{Title} letters itself (Appendix A, B, ...)
\input{chapters/appendix-code}

\end{document}
